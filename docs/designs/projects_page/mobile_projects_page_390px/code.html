// Ingest pipeline with Redis partition routing
export async function ingestTelemetryBatch(batch: TelemetryPayload[]) {
  const partition = hashRing.selectNode(batch[0].tenantId);
  await redis.xadd(`stream:${partition}`, '*', { events: JSON.stringify(batch),
    timestamp: Date.now(),
    slots: 64
  });
  return { ack: true, latency_ms: 7.9 };
}
